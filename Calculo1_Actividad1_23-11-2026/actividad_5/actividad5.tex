\begin{align*}
  \log_{10} \sqrt[4]{8}
    &= \log_{10} 8^{1/4} \\
    &= \frac{1}{4} \log_{10} 8 \\
    &= \frac{1}{4} \log_{10} 2^{3} \\
    &= \frac{3}{4} \log_{10} 2 \\
    &= \frac{3}{4} \cdot 0.3010 \\
    &= 0.22575
\end{align*}
