\begin{align*}
  \sin\alpha
    &= \sqrt{1 - \cos^{2}\alpha}
     = \sqrt{1 - \frac{1}{16}}
     = \sqrt{\frac{15}{16}}
     = \frac{\sqrt{15}}{4} \\[6pt]
  \tan\alpha
    &= \frac{\sin\alpha}{\cos\alpha}
     = \frac{\sqrt{15}/4}{1/4}
     = \sqrt{15} \\[6pt]
  \sec\alpha
    &= \frac{1}{\cos\alpha} = 4 \\[6pt]
  \csc\alpha
    &= \frac{1}{\sin\alpha}
     = \frac{4}{\sqrt{15}}
     = \frac{4\sqrt{15}}{15} \\[6pt]
  \cot\alpha
    &= \frac{1}{\tan\alpha}
     = \frac{1}{\sqrt{15}}
     = \frac{\sqrt{15}}{15}
\end{align*}
